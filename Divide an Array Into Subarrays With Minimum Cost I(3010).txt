class Solution(object):
    def minimumCost(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        
        firstval=nums[0]
        min1=10000
        min2=10000

        for i in range(1,len(nums)):
            if nums[i]<min1:
                min2=min1
                min1=nums[i]
            elif nums[i]<min2:
                min2=nums[i]
        return min1+min2+firstval
