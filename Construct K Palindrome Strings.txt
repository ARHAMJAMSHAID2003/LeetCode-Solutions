class Solution {
public:
    bool canConstruct(string s, int k) {

        
        if(s.length() < k)
            return false;

        
        unordered_map<char,int> freq;
        for(char c : s)
        {
            freq[c]++;
        }

        int oddCount = 0;
        for(auto p : freq)
        {
            if(p.second % 2 != 0)
            {
                oddCount++;
            }
        }

        
        if(oddCount <= k)
            return true;
        else
            return false;
    }
};
