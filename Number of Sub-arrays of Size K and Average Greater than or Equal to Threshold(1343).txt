class Solution(object):
    def numOfSubarrays(self, arr, k, threshold):
        """
        :type arr: List[int]
        :type k: int
        :type threshold: int
        :rtype: int
        """
        count =0
        currsum=0
        for i in range(0,k):
            currsum=currsum+arr[i]

        if currsum/k>=threshold:
            count=count+1
        
        for i in range(1, len(arr)-k+1):
            newsum=currsum-arr[i-1]+arr[i+k-1]
            if newsum/k>=threshold:
                count=count+1
            currsum=newsum
        

        return count
