class Solution(object):
    def appendCharacters(self, s, t):

        i = 0
        j = 0
        vec = []

        while i < len(t) and j < len(s):

            if s[j] != t[i]:
                j += 1       
            else:
                i += 1
                j += 1

        while i < len(t):
            
            vec.append(t[i])
            i += 1

        return len(vec)
