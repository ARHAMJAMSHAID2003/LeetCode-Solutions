class Solution(object):
    def swapNodes(self, head, k):

        curr = head
        count = 0

        while curr is not None:
            count += 1
            curr = curr.next

        count1 = 1
        curr2 = head
        while count1 != k:
            count1 += 1
            curr2 = curr2.next

        val1 = curr2.val

        count2 = 1
        curr3 = head
        while count2 != (count - k) + 1:
            count2 += 1
            curr3 = curr3.next

        curr2.val = curr3.val
        curr3.val = val1

        return head
