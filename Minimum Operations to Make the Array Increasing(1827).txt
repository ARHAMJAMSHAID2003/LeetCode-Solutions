class Solution(object):
    def minOperations(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """ 
        total=0
        
        for i in range(1, len(nums)):
            count=0
            if nums[i] <= nums[i-1]:
                count += (nums[i-1] - nums[i] + 1)
                nums[i]=nums[i-1]+1
            total=total+count          
        return total
