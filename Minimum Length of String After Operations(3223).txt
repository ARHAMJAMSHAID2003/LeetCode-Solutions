class Solution {
public:
    int minimumLength(string s) {


        unordered_map<char,int>store;
        bool match=false;

        for (int i=0;i<s.size();i++)
        {
            if(!match)
            {
                store[s[i]]=1;
                 match=true;
            }

            else
            {
                store[s[i]]+=1;
            }

        }

        int length=s.size();

        for(auto c: store)
        {
            if( c.second>2)
            {
                while(c.second >2)
                {
                length=length-2;
                c.second=c.second-2;
                }
            }


        }


        return length;
        
    }
};
