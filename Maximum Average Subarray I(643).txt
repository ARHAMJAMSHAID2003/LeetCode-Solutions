class Solution(object):
    def findMaxAverage(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: float
        """

        # sum of first k elements
        maxsum = sum(nums[:k])
        totalsum = maxsum

        # slide the window
        for i in range(1, len(nums) - k + 1):
            currsum = maxsum - nums[i - 1] + nums[i + k - 1]
            totalsum = max(currsum, totalsum)
            maxsum = currsum

        return totalsum / float(k)  # ensure floating point division
