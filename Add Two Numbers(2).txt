# Definition for singly-linked list.
# class ListNode(object):
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution(object):
    def addTwoNumbers(self, l1, l2):
        """
        :type l1: Optional[ListNode]
        :type l2: Optional[ListNode]
        :rtype: Optional[ListNode]
        """

        leftover = 0
        curr1 = l1
        curr2 = l2

        sum1 = leftover + curr1.val + curr2.val

        if sum1 >= 10:
            remainder = sum1 % 10
            leftover = sum1 // 10
            head = ListNode(remainder)
        else:
            remainder = sum1
            leftover = 0
            head = ListNode(sum1)

        curr1 = curr1.next
        curr2 = curr2.next
        curr3 = head

        while curr1 is not None and curr2 is not None:

            sum1 = leftover + curr1.val + curr2.val

            if sum1 >= 10:
                remainder = sum1 % 10
                leftover = sum1 // 10
                newnode = ListNode(remainder)
                curr3.next = newnode
                curr3 = curr3.next

            else:
                leftover = 0
                newnode = ListNode(sum1)
                curr3.next = newnode
                curr3 = curr3.next

            curr1 = curr1.next
            curr2 = curr2.next


        while curr1 is not None:

            sum1 = leftover + curr1.val

            if sum1 >= 10:
                remainder = sum1 % 10
                leftover = sum1 // 10
                newnode = ListNode(remainder)

            else:
                leftover = 0
                newnode = ListNode(sum1)

            curr3.next = newnode
            curr3 = curr3.next
            curr1 = curr1.next


        while curr2 is not None:

            sum1 = leftover + curr2.val

            if sum1 >= 10:
                remainder = sum1 % 10
                leftover = sum1 // 10
                newnode = ListNode(remainder)

            else:
                leftover = 0
                newnode = ListNode(sum1)

            curr3.next = newnode
            curr3 = curr3.next
            curr2 = curr2.next


        if leftover > 0:
            curr3.next = ListNode(leftover)

        return head
