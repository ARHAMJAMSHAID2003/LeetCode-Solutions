class Solution(object):
    def getLongestSubsequence(self, words, groups):
        storeindex = []
        values = []

      
        if len(groups) < 2:
            return words
        storeindex.append(0)
        for i in range(1, len(groups)):
            if groups[i] != groups[i - 1]:
                
                   
                    storeindex.append(i)
              

        for i in range( len(storeindex)):
            values.append(words[storeindex[i]])

        return values
